## float_model

Every float64 field has type F64 := fin (x : ℝ) | posInf | negInf | nan, ordered as IEEE 754 orders binary64 (NaN is unordered). Rows P05-20 ('not (x<0) vs 0<=x') and P05-29 ('NaN rejected, +inf accepted') are about NaN and +∞, and neither ℝ nor WithTop ℝ can represent NaN, so those rows would prove nothing. Values are exact reals, so binary64 rounding stays OUT. char[64] is modeled as String (all-NUL = ""). Counts are Nat because every rule compares them to at most 64. t_ns is Int64, lane is Int32, manual_gear_index is Int8.

  | .fin x => .fin (max x 0) | .posInf => .posInf | .negInf => 0 | .nan => 0

/-- IntentFrame part of the committed baseline frame: 06:94 applied to the Pass 1 step 5 rules (06:72). -/
def IntentFrame.committedBaseline (o : OwnView) (t : UInt64) : IntentFrame :=
  { IntentFrame.blank ⟨o.actorId, t⟩ with
    lon := .velocityTarget, vRef := o.vLon.max0, stopAtOdometer := .posInf,
    lat := .laneOffset, targetRoadId := o.roadId, targetLaneId := o.laneId, dRef := o.frenetD,
    signal := .off }

/-- Copy one motion group, with its mode and fields, from src into dst. -/
def IntentFrame.withLonFrom (dst src : IntentFrame) : IntentFrame  -- lon, aRef, vRef, stopAtOdometer, gapTargetActorId, timeGapRef, distanceGapMin
def IntentFrame.withLatFrom (dst src : IntentFrame) : IntentFrame  -- lat, targetRoadId, targetLaneId, dRef, numWaypoints, pathPoints
def IntentFrame.withMotionFrom (dst src : IntentFrame) : IntentFrame :=
  { (dst.withLonFrom src).withLatFrom src with numTrajPoints := src.numTrajPoints, trajectory := src.trajectory }

/-- 10:27, the three cases of the coupled-motion rule. -/
def IntentFrame.coupledMotion (base p s : IntentFrame) : IntentFrame :=
  if s.lon ≠ .none ∧ s.lat ≠ .none then p.withMotionFrom s
  else if s.lon = .none ∧ s.lat = .none then p
  else if s.lon ≠ .none then (p.withMotionFrom base).withLonFrom s
  else (p.withMotionFrom base).withLatFrom s

/-- Group-wise override (10:27: 'the secondary overrides only the groups it requests'). -/
def IntentFrame.groupwise (p s : IntentFrame) : IntentFrame :=
  let q := if s.lon ≠ .none then p.withLonFrom s else p
  let q := if s.lat ≠ .none then q.withLatFrom s else q
  { q with signal := if s.signal ≠ .none then s.signal else p.signal }

/-- A canonical arbiter. The spec does not fix the signal source or the non-STT case; this takes them group-wise. -/
def IntentFrame.arbiterRule (h : Header) (base p s : IntentFrame) : IntentFrame :=
  let m := if p.usesSTT ∨ s.usesSTT then IntentFrame.coupledMotion base p s else p.groupwise s
  ({ m with header := h, signal := (p.groupwise s).signal }).zero

/-- The constraint 10:27 puts on any arbiter output. Outside the STT case it constrains nothing. -/
def IntentFrame.arbiterOK (base p s out : IntentFrame) : Prop :=
  (p.usesSTT ∨ s.usesSTT) →
    let r := IntentFrame.coupledMotion base p s
    out.lon = r.lon ∧ out.lat = r.lat ∧ ∀ fld, r.uses fld = true → out.agreeOn fld r
