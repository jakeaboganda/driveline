```lean
/-! ## Version (09-abi.md:20, driveline_abi.h:12) -/
def encode (major minor : Nat) : Nat := (major <<< 16) ||| (minor <<< 8)
def abiVersion : Nat := 0x00001400   -- DL_ABI_VERSION_0_20
/-- Before 1.0: compatible only when equal (09-abi.md:20). -/
def compatible (component runtime : Nat) : Bool := component == runtime

/-! ## Slice buffer (driveline_abi.h ≈153-169) -/
structure SliceHeader where
  capacity : Nat   -- N
  count : Nat      -- valid entries, ≤ N (≈157)
  entrySize : Nat  -- 8 + sizeof(slice struct)
/-- Ring slot of entry k, k = 0 newest (≈162-163). -/
def slot (cap head k : Nat) : Nat := (head + cap - k) % cap
def entryOffset (h : SliceHeader) (head k : Nat) : Nat := slot h.capacity head k * h.entrySize
/-- The same expression evaluated in `uint32_t`, as C does. -/
def slot32 (cap head k : UInt32) : UInt32 := (head + cap - k) % cap

/-! ## char[N] (09-abi.md:21) -/
abbrev Byte := Fin 256      -- unsigned
abbrev Bytes := List Byte
def fits (n : Nat) (c : Bytes) : Prop := (∀ x ∈ c, x ≠ 0) ∧ c.length + 1 ≤ n
def pad (n : Nat) (c : Bytes) : Bytes := c ++ List.replicate (n - c.length) 0
/-- Content, NUL, zero tail. -/
def wf (n : Nat) (b : Bytes) : Prop := ∃ c, fits n c ∧ b = pad n c
def content (b : Bytes) : Bytes := b.takeWhile (· ≠ 0)
def roadIdN : Nat := 64     -- char road_id[64] (dl_lane_ref_t ≈98 and others)
def paramNameN : Nat := 56  -- char name[56] (dl_param_t ≈341)

/-! ## Output memory (09-abi.md:23) -/
def outOffset (stride i : Nat) : Nat := i * stride
def Disjoint (a la b lb : Nat) : Prop := a + la ≤ b ∨ b + lb ≤ a

/-! ## Header fields and step times (09-abi.md:24-25) -/
inductive OutKind | kinematicState | other
def dtStep (kdiv dtBase : Nat) : Nat := kdiv * dtBase
def stampTime : OutKind → Nat → Nat → Nat
  | .kinematicState, t, dtBase => t + dtBase
  | .other, t, _ => t
structure Frame (β : Type) where
  actorId : Nat
  timestampNs : Nat
  body : β
def finalize (k : OutKind) (id t dtBase : Nat) (f : Frame β) : Frame β :=
  { f with actorId := id, timestampNs := stampTime k t dtBase }

/-! ## Lane sections (09-abi.md:31) -/
variable {α : Type} [LinearOrder α]
/-- Section j covers [s_j, s_{j+1}); the last covers [s_{m-1}, len]. -/
def covers (starts : List α) (len : α) (j : Nat) (s : α) : Prop :=
  ∃ h : j < starts.length, starts[j] ≤ s ∧
    if h' : j + 1 < starts.length then s < starts[j + 1] else s ≤ len

/-! ## world_to_frenet choice (09-abi.md:33) -/
structure Cand (α : Type) where
  roadId : Bytes
  laneId : Int
  s : α
  d : α
  headingDiff : α   -- |psi − driving heading| wrapped to [0, π]
/-- Hint match first (false < true), then heading difference, then road_id
by unsigned bytes, then lane_id as a signed integer. -/
def key (hint : Bytes) (c : Cand α) : Bool ×ₗ α ×ₗ Bytes ×ₗ Int :=
  toLex (c.roadId != hint, toLex (c.headingDiff, toLex (c.roadId, c.laneId)))
def IsChoice (hint : Bytes) (cs : List (Cand α)) (c : Cand α) : Prop :=
  c ∈ cs ∧ ∀ c' ∈ cs, key hint c ≤ key hint c'

/-! ## Successors (09-abi.md:31, 36) -/
abbrev LaneRef := Bytes × Int
def topo (succ : List LaneRef) (maxN : Nat) : List LaneRef × Nat :=
  let sorted := succ.mergeSort (fun a b => decide (toLex a ≤ toLex b))
  (sorted.take maxN, sorted.length)

/-! ## Payloads (07-fmu-packaging.md:15, 17, 19) -/
def le32 (n : Nat) : Bytes :=
  (List.range 4).map fun i => ⟨n / 256 ^ i % 256, Nat.mod_lt _ (by decide)⟩
def de32 (b : Bytes) : Nat := b.foldr (fun x acc => x.val + 256 * acc) 0
/-- Header (four uint32_t, no padding) then count entries, newest first. -/
def serializeBuffer (h : SliceHeader) (entries : List Bytes) : Bytes :=
  le32 h.capacity ++ le32 h.count ++ le32 h.entrySize ++ le32 0 ++ entries.flatten

/-! ## MIME subtype names (07-fmu-packaging.md:25) -/
def kebab : List Char → List Char

  | c :: cs => c.toLower :: cs.flatMap fun x => if x.isUpper then ['-', x.toLower] else [x]
/-- Check this list against §5 and §15; it was taken from the header's struct names. -/
def builtinTypes : List String :=
  ["IntentFrame", "KinematicControlFrame", "ActuatorControlFrame", "KinematicState",
   "VisualSlice", "RadarSlice", "CameraSlice", "SurfaceSlice", "RouteNodes"]
```

**C04** driveline_abi.h ≈162-163. The header is normative (09-abi.md:14). The hypotheses come from the header: head < cap ("slot of the newest entry", ≈166) and k < count ≤ cap (≈157).
```lean
/-- "Entry k (k = 0 newest) starts at entries + ((head + capacity - k) % capacity)
* entry_size" (driveline_abi.h ≈162). -/
theorem slot_zero (h : head < cap) : slot cap head 0 = head
theorem slot_lt (h : head < cap) : slot cap head k < cap
theorem slot_injective (h : head < cap) (h1 : k₁ < cap) (h2 : k₂ < cap)
```

**C05** REFUTE, ring index in `uint32_t` (≈163). `head + capacity` wraps modulo 2^32 when capacity > 2^31.
```lean
/-- REFUTE: (0xBFFFFFFF + 0xC0000000) mod 2^32 = 0x7FFFFFFF, and that value
% 0xC0000000 = 0x7FFFFFFF, not head. -/
theorem refute_slot32 : slot32 0xC0000000 0xBFFFFFFF 0 ≠ 0xBFFFFFFF
/-- With capacity ≤ 2^31 the uint32_t formula is exact. -/
theorem slot32_eq (hc : cap.toNat ≤ 2^31) (h : head < cap) (hk : k ≤ cap) :
    (slot32 cap head k).toNat = slot cap.toNat head.toNat k.toNat
```

## message

world_to_frenet tie-break ends at (road_id, lane_id) and has no s. A lane that contains (X, Y) at two values of s, as on a helical ramp or a self-approaching reference line, leaves s and d undetermined. The same happens at a lane-section boundary where one lane_id continues. Add a final tie-break, for example the smallest s.
* **severity:** medium

**file:** abi/driveline_abi.h

## message

In uint32_t the ring index ((head + capacity - k) % capacity) wraps when capacity > 2^31. Counterexample: cap = 0xC0000000, head = 0xBFFFFFFF, k = 0 gives 0x7FFFFFFF. Bound capacity, or say the arithmetic is done in a wider type. Line number is approximate.
* **severity:** medium

**file:** docs/spec/09-abi.md
