# WP05 statement spec

**Summary:** WP05 spec for §14 and §18, read-only, 5 tool calls. It gives the layout of Driveline.Diagnostics and Driveline.RunRecord, which import and reuse Driveline.Lifecycle, a char-list data model, and a statement for each of the 23 WP05 rows. All 23 row line citations match the source. Five claims are false as written and each gets a counterexample: P11-21 (overlapping static actors get no collision line), P14-06 (teardown reports come after the run error), P14-07 if 'reverse' means reverse tick order, P14-08 (if dl_free_instance can fail, the instance is left Terminated) and P18-01 (-0 and 0 are both written as 0, so encoding is not injective). P18-09 already proposes a counterexample; a witness is supplied. P14-08 duplicates Lifecycle.teardown_frees. Parts of P14-07, P14-08, P14-10, P14-11, P18-01 and P18-05 are deferred to other WPs. Status code -1 is never written as an integer: the report's code member is a dl_status_t name (14:34). Every other integer in the read set is non-negative, so the missing sign rule in 18:16 only matters for the header's files member (§19).

The outcome depends only on the call because no component reads another entity's
output within a tick (11-execution.md:27). -/

  warns : List R
  err : Option R

  | [] => []
  | c :: cs => (out c).warns ++ match (out c).err with | some e => [e] | none => seqReports out cs

  (order.filter ran).flatMap fun c => (out c).warns.map (·, false) ++ (out c).err.toList.map (·, true)

  order.find? fun c => (out c).err.isSome

/-- `checks`: the three steps of 14-diagnostics.md:33 in order. -/

def cstrOk (b : List UInt8) : Bool :=
  match b.findIdx? (· == 0) with
  | some i => (String.fromUTF8? ⟨(b.take i).toArray⟩).isSome
  | none => false
structure Step1View where
  modesOk : Bool
  counts : List Nat
  strs : List (List UInt8)
def step1 (v : Step1View) : Option Code :=
  if v.modesOk && v.counts.all (· ≤ 64) && v.strs.all cstrOk then none else some .errInvalidArg

def dq : Char := Char.ofNat 34
def nl : Char := Char.ofNat 10

def escChar (c : Char) : List Char :=
  if c = dq then [bs, dq] else if c = bs then [bs, bs]
  else if c.toNat < 32 then [bs, 'u', '0', '0', hex (c.toNat / 16), hex (c.toNat % 16)]
  else [c]

def unescape : List Char → List Char   -- left inverse on the image of escape
def jstr (s : List Char) := dq :: escape s ++ [dq]
def jnat (n : Nat) : List Char := Nat.toDigits 10 n

inductive Val (F : Type) | nat (n : Nat) | str (s : List Char) | num (x : F) | raw (j : List Char)
def Val.render (rf : F → List Char) : Val F → List Char
  | .nat n => jnat n | .str s => jstr s | .num x => rf x | .raw j => j
def encodeObj (rf : F → List Char) (ms : List (String × Val F)) : List Char :=
  '{' :: List.intercalate [','] (ms.map fun (k, v) => jstr k.toList ++ ':' :: v.render rf) ++ ['}', nl]

structure Report where
  tick : Nat
  simTimeNs : Nat
  severity : Severity
  code : Code
  inst : List Char
  detail : List Char
structure Collision (F : Type) where
  tick : Nat
  simTimeNs : Nat
  a : Nat
  b : Nat
  vxA : F
  vyA : F
  vxB : F
  vyB : F
inductive Line (F : Type)
  | header (ms : List (String × Val F))   -- `files` as .raw (§19)
  | report (r : Report)
  | collision (c : Collision F)
  | stop (success : Bool) (tick simTimeNs : Nat)   -- the end line
def Line.members : Line F → List (String × Val F)   -- keys in 18:20-23 order; code as .str name
def encode (rf : F → List Char) (l : Line F) := encodeObj rf l.members
def record (hdr : Line F) (body : List (Line F)) (e : Line F) := hdr :: body ++ [e]

/-- Pairs a < b, excluding two static actors, ascending (11-execution.md:24). -/
def pairList (ids : List Nat) (static : Nat → Bool) : List (Nat × Nat) :=
  let s := ids.mergeSort (· ≤ ·)
  s.flatMap fun a => (s.filter fun b => a < b && !(static a && static b)).map (a, ·)
def collisions (contact : Nat → Nat × Nat → Bool) (ps : List (Nat × Nat)) :
    List Nat → List (Nat × Nat) → List (Nat × (Nat × Nat))
  | [], _ => []
  | t :: ts, seen =>
    let new := ps.filter fun p => contact t p && !seen.contains p
    new.map (t, ·) ++ collisions contact ps ts (seen ++ new)

inductive Kind (P : Type) | report (p : P) | collision (a b : Nat)
structure Event (P : Type) where
  tick : Nat
  kind : Kind P
/-- Tick k (11-execution.md:19-24): early at k, pairs at k+1, late at k, window at k+1. -/
structure TickRun (P : Type) where
  early : List P    -- Phases 1-3 and the Phase 4 commit
  pairs : List (Nat × Nat)
  late : List P     -- terminate when, on
  window : List P   -- splice window, including its teardown calls (18:25)
def TickRun.events (k : Nat) (t : TickRun P) : List (Event P)
structure Run (P : Type) where
  cold : List P
  spawnPairs : List (Nat × Nat)
  ticks : List (TickRun P)   -- index = k
  ending : Ending
  teardown : List P
def Run.pre (r : Run P) : List (Event P)   -- cold@0, spawnPairs@0, then ticks[k].events k
def Run.body (r : Run P) := r.pre ++ r.teardown.map fun p => ⟨teardownTick r.ending, .report p⟩
def Run.WF (r : Run P) : Prop := match r.ending with
  | .failed .coldInit _ => r.ticks = [] ∧ r.spawnPairs = []
  | .failed (.exec k) _ | .failed (.splice k) _ | .succeeded k => r.ticks.length = k + 1
def lastCommitted : Ending → Option Nat
  | .failed .coldInit _ => none | .failed (.exec k) c => some (if c then k + 1 else k)
  | .failed (.splice k) _ => some (k + 1) | .succeeded k => some (k + 1)
def endTick (e : Ending) : Nat := (lastCommitted e).getD 0
