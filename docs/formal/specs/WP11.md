structure Body (α : Type) where
  x : α
  y : α
  z : α
  c : α   -- cos yaw
  s : α   -- sin yaw
  xlo : α
  xhi : α
  ylo : α
  yhi : α
  H : α

variable {α : Type} [Add α] [Mul α] [Neg α] [LT α] [LE α] [Max α] [Min α]

def u (b : Body α) : α × α := (b.c, b.s)
def n (b : Body α) : α × α := (-b.s, b.c)
/-- 11:24 "p + x u + y n ... evaluated left to right per component" -/
def corner (b : Body α) (ex ey : α) : α × α :=
  (b.x + ex * b.c + ey * -b.s, b.y + ex * b.s + ey * b.c)
/-- 11:24 "q_X w_X + q_Y w_Y" -/
def proj (w q : α × α) : α := q.1 * w.1 + q.2 * w.2
def pmax (b : Body α) (w : α × α) : α :=
  max (max (proj w (corner b b.xlo b.ylo)) (proj w (corner b b.xhi b.ylo)))
      (max (proj w (corner b b.xhi b.yhi)) (proj w (corner b b.xlo b.yhi)))
def pmin (b : Body α) (w : α × α) : α :=   -- same with min
  min (min (proj w (corner b b.xlo b.ylo)) (proj w (corner b b.xhi b.ylo)))
      (min (proj w (corner b b.xhi b.yhi)) (proj w (corner b b.xlo b.yhi)))
def axes (a b : Body α) : List (α × α) := [u a, n a, u b, n b]
def Separated (a b : Body α) (w : α × α) : Prop := pmax a w < pmin b w ∨ pmax b w < pmin a w
def FootprintContact (a b : Body α) : Prop := ∀ w ∈ axes a b, ¬ Separated a b w
/-- closed intervals [z, z+H] intersect; the self conjuncts make an empty interval (H < 0) never touch -/
def HeightContact (a b : Body α) : Prop :=
  a.z ≤ a.z + a.H ∧ b.z ≤ b.z + b.H ∧ a.z ≤ b.z + b.H ∧ b.z ≤ a.z + a.H
def Contact (a b : Body α) : Prop := FootprintContact a b ∧ HeightContact a b

/-- Tier 0 body-frame box (03:20 Placement; objects 03:42) -/
structure Geom where
  xlo : ℝ
  xhi : ℝ
  ylo : ℝ
  yhi : ℝ
  H : ℝ
def Geom.ofTier0 (L oF oR W Hb : ℝ) : Geom := ⟨-oR, L + oF, -(W / 2), W / 2, Hb⟩  -- y bounds: finding 1

structure Actor where
  x : ℝ
  y : ℝ
  z : ℝ
  ψ : ℝ
  g : Geom
noncomputable def Actor.body (a : Actor) : Body ℝ :=
  ⟨a.x, a.y, a.z, Real.cos a.ψ, Real.sin a.ψ, a.g.xlo, a.g.xhi, a.g.ylo, a.g.yhi, a.g.H⟩
/-- the closed footprint; uIcc equals Icc under positive dimensions and needs no added hypothesis -/
def rect (b : Body ℝ) : Set (ℝ × ℝ) :=
  {q | ∃ ex ∈ Set.uIcc b.xlo b.xhi, ∃ ey ∈ Set.uIcc b.ylo b.yhi, q = corner b ex ey}
noncomputable def Actor.move (φ tx ty tz : ℝ) (a : Actor) : Actor :=
  { a with x := tx + a.x * Real.cos φ - a.y * Real.sin φ, y := ty + a.x * Real.sin φ + a.y * Real.cos φ,
           z := a.z + tz, ψ := a.ψ + φ }
open Classical in
noncomputable def contactAt (w : ℕ → ℕ → Actor) (t : ℕ) (p : ℕ × ℕ) : Bool :=
  decide (Contact (w t p.1).body (w t p.2).body)
def sq (x z : ℝ) : Actor := ⟨x, 0, z, 0, ⟨0, 1, 0, 1, 1⟩⟩

## proof_hint

(←) For q ∈ rect, proj w q = proj w p + ex⟨u,w⟩ + ey⟨n,w⟩ is affine on a uIcc box, so pmin ≤ proj w q ≤ pmax; a shared point rules out strict separation. (→) Use cos²+sin²=1 to prove rect b = ⋂ of 4 support half-planes {proj w q ≥ pmin b w}, {proj w q ≤ pmax b w} for w ∈ {u b, n b}. Then rect a ∩ rect b is an intersection of 8 convex sets in ℝ×ℝ (finrank 2). By Helly it is nonempty if every 3 of them meet. Three half-planes from one rectangle contain that rectangle, which is nonempty. Two from a and H3 from b: if they don't meet, rect a ∩ H3 = ∅, so every corner of a is strictly outside H3. That gives pmax a w < pmin b w, or pmax b w < pmin a w, with w ∈ axes, which contradicts FootprintContact. The 1-from-a, 2-from-b case is symmetric. This proof needs no hypothesis on dimensions.
