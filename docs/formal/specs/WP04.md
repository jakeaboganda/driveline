abbrev Tick := ℕ
def tickTime (dt : ℕ+) (k : Tick) : ℕ := k * dt
def validRate (dt : ℕ+) (f : ℚ) : Prop := ∃ k : ℕ+, (k : ℚ) * dt * f = 10 ^ 9
def defaultDiv : ℕ+ := 1
def runs (d : ℕ+) (k : Tick) : Prop := k % d = 0          -- literal 11:15; Decidable
def lastRun (d : ℕ+) (k : Tick) : Tick := d * (k / d)
def held {α} (d : ℕ+) (out : Tick → α) (k : Tick) : α := out (lastRun d k)  -- ZOH
def stepDt (d dt : ℕ+) : ℕ := d * dt
def tFirst (d : ℕ+) (k : Tick) : Tick := d * ((k + d - 1) / d)  -- least run tick ≥ k

-- Phase 2 executor: one buffer slot per component, components in `order`.
structure P2 (ι α : Type*) where
  order : List ι
  div   : ι → ℕ+
  step  : ι → Tick → (ι → α) → α   -- reads the buffer as it stands at its turn
variable {ι α : Type*} [DecidableEq ι]
def turn (p : P2 ι α) (k : Tick) (b : ι → α) (i : ι) : ι → α :=
  if runs (p.div i) k then Function.update b i (p.step i k b) else b
def phase2 (p : P2 ι α) (k : Tick) (b : ι → α) := p.order.foldl (turn p k) b
def pre (p : P2 ι α) (b0 : ι → α) : Tick → (ι → α)
  | 0 => b0 | k + 1 => phase2 p k (pre p b0 k)
def after (p : P2 ι α) (b0) (k : Tick) := phase2 p k (pre p b0 k)
def seenBy (p : P2 ι α) (b0) (k : Tick) (c : ι) : ι → α :=
  (p.order.takeWhile (· ≠ c)).foldl (turn p k) (pre p b0 k)

inductive Phase | sensors | control | physics | commit deriving DecidableEq, Repr
def Phase.idx : Phase → ℕ | .sensors => 1 | .control => 2 | .physics => 3 | .commit => 4
def phases (k : Tick) : List Phase :=
  if k = 0 then [.control, .physics, .commit] else [.sensors, .control, .physics, .commit]
inductive P4Step | commit | contact | terminate | onStatements deriving DecidableEq, Repr
/-- Phase 4 steps and the `on` ids that fire. -/
def phase4 (term : Bool) (onConds : List (ℕ × Bool)) : List P4Step × List ℕ :=
  if term then ([.commit, .contact, .terminate], [])
  else ([.commit, .contact, .terminate, .onStatements], (onConds.filter (·.2)).map (·.1))
def executed (term : Tick → Bool) (k : Tick) : Prop := ∀ j < k, term j = false
def predTime (dt : ℕ+) (k : Tick) : ℕ := tickTime dt k + dt

structure KState (σ : Type*) where
  timestamp_ns : ℕ
  rest : σ                          -- every other field
def commitStatic {σ} (s : KState σ) (t' : ℕ) : KState σ := { s with timestamp_ns := t' }
structure Pose where (x y : ℝ)
noncomputable def odoStep (o : ℝ) (p p' : Pose) : ℝ :=
  o + Real.sqrt ((p'.x - p.x) ^ 2 + (p'.y - p.y) ^ 2)

structure Entity where (ids : Finset ℕ) (ne : ids.Nonempty)   -- actor = singleton
def Entity.key (e : Entity) : ℕ := e.ids.min' e.ne

structure Chain (n : ℕ) where (path : Fin n → Fin n → Prop)   -- transitive closure of data edges
def Chain.IsDAG (c : Chain n) : Prop := (∀ a, ¬ c.path a a) ∧ ∀ a b z, c.path a b → c.path b z → c.path a z
def tieOrder (c : Chain n) (a b : Fin n) : Prop :=
  c.path a b ∨ (¬ c.path a b ∧ ¬ c.path b a ∧ a < b)

def groupOrder {κ} (cs : List κ) (bs : List ℕ) : List (κ × ℕ) := cs.flatMap fun c => bs.map (c, ·)
def Precedes {β} (l : List β) (x y : β) : Prop := ∃ i j, i < j ∧ l[i]? = some x ∧ l[j]? = some y

def block {E S β} [DecidableEq E] (f : E → S → β → β) (s : S) (st : E → β) (e : E) : E → β :=

### FmuTime.lean: rounding decision

Rounding is abstract: `rnd fl : ℚ → ℚ`, with no axioms. IEEE 754 addition returns the correctly rounded exact sum, so `fl (a + b)` is binary64 `a + b` when `fl` is round-to-nearest-even. Contiguity (07:36) is a property of the recurrence. Proving it for every `fl` is stronger than proving it for binary64 and needs no float model. Taking `rnd = fl = id` gives the exact-arithmetic check that the points are the tick times.
```lean
structure Times where (rnd fl : ℚ → ℚ)
def startTime (T : Times) (tNs : ℕ) : ℚ := T.rnd ((tNs : ℚ) / 10 ^ 9)
def stepSize  (T : Times) (hNs : ℕ) : ℚ := T.rnd ((hNs : ℚ) / 10 ^ 9)
def point (T : Times) (tNs hNs : ℕ) : ℕ → ℚ
  | 0 => startTime T tNs
  | j + 1 => T.fl (point T tNs hNs j + stepSize T hNs)

inductive FmiStatus | ok | warning | discard | error | fatal deriving DecidableEq, Repr
def doStepOk (s : FmiStatus) (terminateSimulation earlyReturn : Bool) : Bool :=
  (s == .ok || s == .warning) && !terminateSimulation && !earlyReturn

structure FmiHistory where (sawError sawFatal : Bool)  -- some FMI call returned fmi3Error / fmi3Fatal
/-- FMI Step Mode by the 07:57-61 mapping: only after dl_exit_init_mode. -/
def fmiStepMode : Lifecycle.State → Bool | .stepMode => true | _ => false
inductive FmiCall | terminate | freeInstance deriving DecidableEq, Repr
def termCalls (s : Lifecycle.State) (h : FmiHistory) : List FmiCall :=
  if fmiStepMode s && !h.sawError && !h.sawFatal then [.terminate] else []
def freeCalls (h : FmiHistory) : List FmiCall := if h.sawFatal then [] else [.freeInstance]

structure Fmu (X U Y : Type*) where (F : X → U → ℚ → ℚ → X) (H : X → Y)
def Fmu.state (m : Fmu X U Y) (x0 : X) (u : ℕ → U) (p : ℕ → ℚ) (h : ℚ) : ℕ → X
  | 0 => x0 | j + 1 => m.F (m.state x0 u p h j) (u j) (p j) h
/-- Output read after the j-th fmi3DoStep: the FMU at p j + h. -/
def Fmu.out (m : Fmu X U Y) x0 u p h (j : ℕ) : Y := m.H (m.state x0 u p h (j + 1))
```

## Findings

* **id:** F1

**severity:** high

**location:** docs/spec/11-execution.md:27

## text

Read pair by pair, the tie-break 'components that have no data path between them in the left-to-right order of their calls' can form a cycle. With A(0), B(1), C(2) and only C→A, the rule gives A<B, B<C and C<A. This is consistent only if the chain grammar guarantees that every call positioned between X and Y has a path to X whenever Y→X. I could not check this within the tool budget. Suggested fix: 'in the topological order that at each step runs the leftmost component whose upstreams have all run'.
