open Real Matrix
def e₁ : Fin 3 → ℝ := ![1, 0, 0]   -- +X East / +x forward
def e₂ : Fin 3 → ℝ := ![0, 1, 0]   -- +Y North / +y left
def e₃ : Fin 3 → ℝ := ![0, 0, 1]   -- +Z Up
noncomputable def Rx (φ : ℝ) : Matrix (Fin 3) (Fin 3) ℝ := !![1, 0, 0; 0, cos φ, -sin φ; 0, sin φ, cos φ]
noncomputable def Ry (θ : ℝ) : Matrix (Fin 3) (Fin 3) ℝ := !![cos θ, 0, sin θ; 0, 1, 0; -sin θ, 0, cos θ]
noncomputable def Rz (ψ : ℝ) : Matrix (Fin 3) (Fin 3) ℝ := !![cos ψ, -sin ψ, 0; sin ψ, cos ψ, 0; 0, 0, 1]
/-- Body-to-World rotation. -/
noncomputable def eulerZYX (ψ θ φ : ℝ) : Matrix (Fin 3) (Fin 3) ℝ := Rz ψ * Ry θ * Rx φ
def skew (u : Fin 3 → ℝ) : Matrix (Fin 3) (Fin 3) ℝ := !![0, -u 2, u 1; u 2, 0, -u 0; -u 1, u 0, 0]
/-- Rodrigues: rotation by `a` about the unit axis `u`. -/
noncomputable def axisRot (u : Fin 3 → ℝ) (a : ℝ) : Matrix (Fin 3) (Fin 3) ℝ := 1 + sin a • skew u + (1 - cos a) • (skew u * skew u)
/-- 02:19 the World frame rotated by ψ about +Z. -/
noncomputable def headingFrame (ψ : ℝ) : Matrix (Fin 3) (Fin 3) ℝ := Rz ψ
/-- 04:37 ANGLE interpolation. -/
noncomputable def interpAngle (α vk vk1 : ℝ) : ℝ := wrap (vk1 + α * wrap (vk - vk1))

open Real Matrix
noncomputable def rot2 (a : ℝ) : Matrix (Fin 2) (Fin 2) ℝ := !![cos a, -sin a; sin a, cos a]
def vyCg (lr vLat r : ℝ) : ℝ := vLat + lr * r
def sgn0 (x : ℝ) : ℝ := if 0 ≤ x then 1 else -1
noncomputable def atan2 (y x : ℝ) : ℝ := Complex.arg ⟨x, y⟩
noncomputable def betaCg (lr vLon vLat r : ℝ) : ℝ := atan2 (sgn0 vLon * vyCg lr vLat r) |vLon|
/-- 05:122 no Tier 1 → β = 0. -/
noncomputable def betaReported : Option ℝ → ℝ → ℝ → ℝ → ℝ
  | none, _, _, _ => 0
  | some lr, v, vl, r => betaCg lr v vl r
noncomputable def ksYawRate (L vLon δ : ℝ) : ℝ := vLon / L * tan δ
/-- 05:76; `between` is the component's interpolation. -/
def trajTarget {α : Type} (time : α → ℤ) (between : α → α → ℤ → α) : List α → ℤ → Option α
  | [], _ => none
  | [p], _ => some p
  | p :: q :: rest, t => if t ≤ time p then some p else if t < time q then some (between p q t) else trajTarget time between (q :: rest) t
noncomputable def odometer (X Y : ℕ → ℝ) : ℕ → ℝ
  | 0 => 0
  | k + 1 => odometer X Y k + √((X (k+1) - X k) * (X (k+1) - X k) + (Y (k+1) - Y k) * (Y (k+1) - Y k))
